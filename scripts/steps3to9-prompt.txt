Parse training/narration-script.txt. For each [SECTION: name] ... [END] block run:

Then combine all audio:
  FFMPEG="C:\\Users\\digvi\\AppData\\Local\\Microsoft\\WinGet\\Links\\ffmpeg.exe"
  ls training/audio/section-*.mp3 | while read f; do echo "file '$(cygpath -w "$f")'"; done > training/audio/filelist.txt
  "$FFMPEG" -y -f concat -safe 0 -i "$(cygpath -w training/audio/filelist.txt)" -acodec libmp3lame "$(cygpath -w training/audio/combined.mp3)"
